cmake_minimum_required(VERSION 3.24)
if(DEFINED SDK_PARENT_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "${SDK_PARENT_DEPLOYMENT_TARGET}" CACHE STRING "Parent SDK fixture policy" FORCE)
endif()
project(LoopZConsumerFixture LANGUAGES C CXX)
include(CTest)
add_custom_target(format)
add_custom_target(format-check)
add_test(NAME consumer-parent-sentinel COMMAND "${CMAKE_COMMAND}" -E true)
if(NOT LOOPZ_SOURCE OR NOT SHARED_ZCORE_SOURCE)
  message(FATAL_ERROR "Set LOOPZ_SOURCE and SHARED_ZCORE_SOURCE to pinned source checkouts")
endif()
set(ZCORE_BUILD_MEDIA OFF CACHE BOOL "" FORCE)
set(ZCORE_BUILD_ANALYSIS OFF CACHE BOOL "" FORCE)
set(ZCORE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZCORE_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SHARED_ZCORE_SOURCE}" zcore)
# Resetting overlay options after target creation must not hide actual sources.
if(SDK_RESET_OVERLAY_CACHE)
  set(ZCORE_GRAPH_RUNNER_SOURCE "${SHARED_ZCORE_SOURCE}/zdsp/src/runtime/graph_runner.cpp" CACHE FILEPATH "" FORCE)
  set(ZCORE_BUILTIN_NODES_SOURCE "${SHARED_ZCORE_SOURCE}/zdsp/src/runtime/builtin_nodes.cpp" CACHE FILEPATH "" FORCE)
endif()
if(SDK_REMOVE_ROUTING_KERNEL)
  get_target_property(_routing_sources zcore_routing SOURCES)
  list(FILTER _routing_sources EXCLUDE REGEX "cached_fir_kernels[.]cpp$")
  set_property(TARGET zcore_routing PROPERTY SOURCES "${_routing_sources}")
endif()
if(SDK_ROUTING_OVERLAY)
  target_sources(zcore_routing PRIVATE "${SDK_ROUTING_OVERLAY}")
endif()
set(_deployment_before_loopz "${CMAKE_OSX_DEPLOYMENT_TARGET}")
set(LOOPZ_BUILD_MACOS_BACKEND OFF CACHE BOOL "" FORCE)
set(LOOPZ_BUILD_ENGINE OFF CACHE BOOL "" FORCE)
if(SDK_CONTROL_CONSUMER)
  set(LOOPZ_BUILD_RUNTIME_LIBRARY OFF CACHE BOOL "" FORCE)
else()
  set(LOOPZ_BUILD_RUNTIME_LIBRARY ON CACHE BOOL "" FORCE)
endif()
set(LOOPZ_BUILD_CONTROL_LIBRARY "${SDK_CONTROL_CONSUMER}" CACHE BOOL "" FORCE)
set(LOOPZ_BUILD_LINUX_DESKTOP OFF CACHE BOOL "" FORCE)
set(LOOPZ_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(NOT DEFINED LOOPZ_SHARED_ZCORE_ROOT)
  set(LOOPZ_SHARED_ZCORE_ROOT "${SHARED_ZCORE_SOURCE}")
endif()
add_subdirectory("${LOOPZ_SOURCE}" loopz)
if(NOT "${CMAKE_OSX_DEPLOYMENT_TARGET}" STREQUAL "${_deployment_before_loopz}")
  message(FATAL_ERROR "LoopZ changed the parent's deployment target policy")
endif()
if(SDK_POST_INCLUDE_OVERLAY)
  target_sources(zdsp_runtime PRIVATE "${SDK_POST_INCLUDE_OVERLAY}")
endif()
if(SDK_POST_INCLUDE_ROUTING_OVERLAY)
  target_sources(zcore_routing PRIVATE "${SDK_POST_INCLUDE_ROUTING_OVERLAY}")
endif()
foreach(unexpected IN ITEMS loopz-engine loopzctl loopz_control loopz-document-json-tests loopz-standalone-audio-tests)
  if(TARGET "${unexpected}")
    message(FATAL_ERROR "Library-only embedding created unexpected target ${unexpected}")
  endif()
endforeach()
if(NOT BUILD_TESTING)
  message(FATAL_ERROR "LoopZ changed the parent's testing policy")
endif()
add_executable(loopz-sdk-consumer main.cpp)
target_link_libraries(loopz-sdk-consumer PRIVATE LoopZ::Runtime)
add_test(NAME consumer-native-source COMMAND loopz-sdk-consumer)

if(SDK_CONTROL_CONSUMER)
  add_executable(loopz-control-sdk-consumer control_main.cpp)
  target_link_libraries(loopz-control-sdk-consumer PRIVATE LoopZ::Control)
  add_test(NAME consumer-control-restore COMMAND loopz-control-sdk-consumer "${CMAKE_CURRENT_BINARY_DIR}/control-state.json")
elseif(TARGET loopz_control_core)
  message(FATAL_ERROR "Runtime-only embedding created unexpected Control target")
endif()
