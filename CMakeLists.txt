cmake_minimum_required(VERSION 3.24)
project(LoopZ VERSION 0.1.0 LANGUAGES C CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "The feasibility build targets macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "14.2" CACHE STRING "Minimum macOS")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
add_custom_target(format
  COMMAND python3 "${PROJECT_SOURCE_DIR}/scripts/format_code.py"
  WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
  COMMENT "Formatting LoopZ sources")
add_custom_target(format-check
  COMMAND python3 "${PROJECT_SOURCE_DIR}/scripts/format_code.py" --check
  WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
  COMMENT "Checking LoopZ source formatting")
find_package(Threads REQUIRED)
add_subdirectory(cmake/singz)
add_library(loopz_core STATIC native/core/graph.cpp native/core/capture_buffer.cpp)
target_include_directories(loopz_core PUBLIC native/include)
target_link_libraries(loopz_core PUBLIC SingZ::zdsp_runtime)
target_compile_options(loopz_core PRIVATE -Wall -Wextra -Werror)
add_library(loopz_bridge STATIC native/platform/bridge_client.mm)
target_include_directories(loopz_bridge PUBLIC native/include)
target_compile_options(loopz_bridge PRIVATE -fblocks -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(loopz_bridge PUBLIC "-framework Foundation")
add_executable(loopz-broker native/broker/main.mm)
target_include_directories(loopz-broker PRIVATE native/include)
target_compile_options(loopz-broker PRIVATE -fblocks -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(loopz-broker PRIVATE "-framework Foundation" "-framework SystemConfiguration")
add_library(LoopZAudio MODULE native/driver/LoopZAudio.c native/driver/driver_bridge.cpp)
target_include_directories(LoopZAudio PRIVATE native/include native/driver)
target_compile_options(LoopZAudio PRIVATE -fblocks)
target_link_libraries(LoopZAudio PRIVATE loopz_bridge "-framework CoreAudio" "-framework CoreFoundation")
set_target_properties(LoopZAudio PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION driver PREFIX ""
  MACOSX_BUNDLE_INFO_PLIST "${PROJECT_SOURCE_DIR}/packaging/driver.plist")
add_executable(loopz-engine MACOSX_BUNDLE native/engine/main.mm native/platform/capture.mm native/platform/bridge_status.mm)
target_include_directories(loopz-engine PRIVATE native/include)
target_compile_options(loopz-engine PRIVATE -fblocks -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(loopz-engine PRIVATE loopz_core loopz_bridge SingZ::zcore_device
  "-framework Foundation" "-framework AppKit" "-framework AVFoundation" "-framework CoreAudio")
set_target_properties(loopz-engine PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${PROJECT_SOURCE_DIR}/packaging/engine.plist"
  OUTPUT_NAME "LoopZ Engine")
add_executable(loopz-probe native/probe/main.mm)
target_include_directories(loopz-probe PRIVATE native/include)
target_compile_options(loopz-probe PRIVATE -fblocks -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(loopz-probe PRIVATE loopz_core "-framework Foundation" "-framework CoreAudio" "-framework AudioToolbox")
add_executable(loopz-test-source MACOSX_BUNDLE native/probe/main.mm)
target_include_directories(loopz-test-source PRIVATE native/include)
target_compile_options(loopz-test-source PRIVATE -fblocks -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(loopz-test-source PRIVATE "-framework Foundation" "-framework CoreAudio" "-framework AudioToolbox")
set_target_properties(loopz-test-source PROPERTIES OUTPUT_NAME "LoopZ Test Source"
  MACOSX_BUNDLE_INFO_PLIST "${PROJECT_SOURCE_DIR}/packaging/test-source.plist")
if(BUILD_TESTING)
  add_executable(loopz-core-tests tests/core_tests.cpp)
  target_link_libraries(loopz-core-tests PRIVATE loopz_core Threads::Threads)
  target_compile_options(loopz-core-tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME loopz-core COMMAND loopz-core-tests)
  add_test(NAME vendor-integrity COMMAND python3 "${PROJECT_SOURCE_DIR}/scripts/verify_vendor.py")
  add_executable(loopz-driver-tests tests/driver_tests.cpp)
  target_link_libraries(loopz-driver-tests PRIVATE "-framework CoreAudio" "-framework CoreFoundation")
  target_compile_options(loopz-driver-tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_dependencies(loopz-driver-tests LoopZAudio)
  add_test(NAME loopz-driver COMMAND loopz-driver-tests "$<TARGET_FILE:LoopZAudio>")
endif()
if(BUILD_TESTING)
  add_executable(loopz-probe-tests tests/probe_tests.cpp)
  target_include_directories(loopz-probe-tests PRIVATE native/include)
  target_compile_options(loopz-probe-tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME loopz-probe-signal COMMAND loopz-probe-tests)
endif()
add_executable(loopz-install-check native/install/check.mm)
target_include_directories(loopz-install-check PRIVATE native/include)
target_compile_options(loopz-install-check PRIVATE -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(loopz-install-check PRIVATE "-framework Foundation" "-framework CoreAudio" "-framework AudioToolbox")
if(BUILD_TESTING)
  add_test(NAME loopz-installer COMMAND python3 "${PROJECT_SOURCE_DIR}/tests/install_tests.py")
endif()
